```ts
export const emailsQueryKey = ["emails"] as const;

export const emailsQueryOptions = {
	queryKey: emailsQueryKey,
	queryFn: identityApi.listEmails,
	retry: false,
};

export function useEmails() {
	return useQuery(emailsQueryOptions);
}

export function useAddEmail() {
	const qc = useQueryClient();
	return useMutation({
		mutationFn: (input: AddEmailInput) => identityApi.addEmail(input),
		onSuccess: () => qc.invalidateQueries({ queryKey: emailsQueryKey }),
	});
}

export function useSetPrimaryEmail() {
	const qc = useQueryClient();
	return useMutation({
		mutationFn: ({ id, input }: { id: number; input: SetPrimaryInput }) =>
			identityApi.setPrimaryEmail(id, input),
		onSuccess: () => {
			qc.invalidateQueries({ queryKey: emailsQueryKey });
			qc.invalidateQueries({ queryKey: meQueryKey });
		},
	});
}

export function useRemoveEmail() {
	const qc = useQueryClient();
	return useMutation({
		mutationFn: (id: number) => identityApi.removeEmail(id),
		onSuccess: () => qc.invalidateQueries({ queryKey: emailsQueryKey }),
	});
}
```

Create `src/features/identity/components/AddEmailForm.tsx`:

```tsx
import { zodResolver } from "@hookform/resolvers/zod";
import { useForm } from "react-hook-form";
import { useTranslation } from "react-i18next";
import { Alert, AlertDescription, Button, Field, Input } from "@/ui";
import { parseApiError } from "../api";
import { useAddEmail } from "../queries";
import { type AddEmailInput, addEmailSchema } from "../schemas";

export function AddEmailForm({
	onSuccess,
}: {
	onSuccess: (email: string) => void;
}) {
	const { t } = useTranslation();
	const add = useAddEmail();
	const {
		register,
		handleSubmit,
		setError,
		formState: { errors, isSubmitting },
	} = useForm<AddEmailInput>({
		resolver: zodResolver(addEmailSchema),
		defaultValues: { email: "", current_password: "" },
	});

	async function onSubmit(values: AddEmailInput) {
		try {
			await add.mutateAsync(values);
			onSuccess(values.email);
		} catch (error) {
			const parsed = parseApiError(error);
			for (const [field, message] of Object.entries(parsed.fieldErrors)) {
				setError(field as keyof AddEmailInput, { message });
			}
			if (parsed.throttled) {
				setError("root.server", { message: t("account.throttled") });
			} else if (parsed.message) {
				setError("root.server", { message: parsed.message });
			}
		}
	}

	return (
		<form
			onSubmit={handleSubmit(onSubmit)}
			className="flex flex-col gap-4"
			noValidate
		>
			<Field
				id="new_email"
				label={t("account.newEmail")}
				error={errors.email?.message}
			>
				<Input type="email" autoComplete="email" {...register("email")} />
			</Field>
			<Field
				id="add_email_password"
				label={t("account.currentPassword")}
				error={errors.current_password?.message}
			>
				<Input
					type="password"
					autoComplete="current-password"
					{...register("current_password")}
				/>
			</Field>
			{errors.root?.server ? (
				<Alert variant="destructive">
					<AlertDescription>{errors.root.server.message}</AlertDescription>
				</Alert>
			) : null}
			<Button type="submit" disabled={isSubmitting}>
				{t("account.addEmailSubmit")}
			</Button>
		</form>
	);
}
```

Create `src/features/identity/components/SetPrimaryDialog.tsx` (controlled `open` state — each instance owns its own state, so multiple dialogs on the page don't interfere):

```tsx
import { zodResolver } from "@hookform/resolvers/zod";
import { Dialog } from "radix-ui";
import { useState } from "react";
import { useForm } from "react-hook-form";
import { useTranslation } from "react-i18next";
import { Alert, AlertDescription, Button, Field, Input } from "@/ui";
import { parseApiError } from "../api";
import { useSetPrimaryEmail } from "../queries";
import {
	type EmailAddress,
	type SetPrimaryInput,
	setPrimarySchema,
} from "../schemas";

export function SetPrimaryDialog({ address }: { address: EmailAddress }) {
	const { t } = useTranslation();
	const [open, setOpen] = useState(false);
	const setPrimary = useSetPrimaryEmail();
	const {
		register,
		handleSubmit,
		setError,
		reset,
		formState: { errors, isSubmitting },
	} = useForm<SetPrimaryInput>({
		resolver: zodResolver(setPrimarySchema),
		defaultValues: { current_password: "" },
	});

	async function onSubmit(values: SetPrimaryInput) {
		try {
			await setPrimary.mutateAsync({ id: address.id, input: values });
			reset();
			setOpen(false);
		} catch (error) {
			const parsed = parseApiError(error);
			if (parsed.fieldErrors.current_password) {
				setError("current_password", {
					message: parsed.fieldErrors.current_password,
				});
			} else {
				setError("root.server", {
					message: parsed.message ?? t("account.genericError"),
				});
			}
		}
	}

	return (
		<Dialog.Root open={open} onOpenChange={setOpen}>
			<Dialog.Trigger asChild>
				<Button variant="outline" size="sm">
					{t("account.makePrimary")}
				</Button>
			</Dialog.Trigger>
			<Dialog.Portal>
				<Dialog.Overlay className="fixed inset-0 z-40 bg-black/40" />
				<Dialog.Content className="fixed start-1/2 top-1/2 z-50 w-[min(90vw,24rem)] -translate-x-1/2 -translate-y-1/2 rounded-lg border border-border bg-background p-6 shadow-lg rtl:translate-x-1/2">
					<Dialog.Title className="font-display text-lg font-semibold">
						{t("account.makePrimaryTitle")}
					</Dialog.Title>
					<Dialog.Description className="mt-1 text-sm text-muted-foreground">
						{t("account.makePrimaryBody", { email: address.email })}
					</Dialog.Description>
					<form
						onSubmit={handleSubmit(onSubmit)}
						className="mt-4 flex flex-col gap-4"
						noValidate
					>
						<Field
							id="primary_password"
							label={t("account.currentPassword")}
							error={errors.current_password?.message}
						>
							<Input
								type="password"
								autoComplete="current-password"
								{...register("current_password")}
							/>
						</Field>
						{errors.root?.server ? (
							<Alert variant="destructive">
								<AlertDescription>
									{errors.root.server.message}
								</AlertDescription>
							</Alert>
						) : null}
						<div className="flex justify-end gap-2">
							<Dialog.Close asChild>
								<Button type="button" variant="outline">
									{t("account.cancel")}
								</Button>
							</Dialog.Close>
							<Button type="submit" disabled={isSubmitting}>
								{t("account.confirm")}
							</Button>
						</div>
					</form>
				</Dialog.Content>
			</Dialog.Portal>
		</Dialog.Root>
	);
}
```

Closing: the dialog is controlled by local `open` state — success calls `setOpen(false)`; Cancel/overlay/Escape close via radix's `onOpenChange`. Because each `SetPrimaryDialog` instance has its own state, the multiple dialogs the `EmailAddresses` card renders never interfere.

```tsx
import { useState } from "react";
import { useTranslation } from "react-i18next";
import {
	Alert,
	AlertDescription,
	Button,
	Card,
	CardContent,
	CardHeader,
	CardTitle,
	Spinner,
} from "@/ui";

import { AddEmailForm } from "./AddEmailForm";
import { RemoveEmailDialog } from "./RemoveEmailDialog";
import { SetPrimaryDialog } from "./SetPrimaryDialog";

export function EmailAddresses() {

	const { data: emails, isPending, isError } = useEmails();

	const [adding, setAdding] = useState(false);

								<span className="me-auto text-sm">{address.email}</span>

								) : null}

										size="sm"

											setNotice(t("account.resent"));
										}}
									>
										{t("account.resend")}
									</Button>
								) : null}
								{!address.primary ? <RemoveEmailDialog address={address} /> : null}
							</li>
						))}
					</ul>
				) : null}
				{notice ? (
					<Alert>
						<AlertDescription>{notice}</AlertDescription>
					</Alert>

					<AddEmailForm
						onSuccess={(email) => {
							setAdding(false);
							setNotice(t("account.verificationSent", { email }));

						{t("account.addEmail")}
					</Button>
```

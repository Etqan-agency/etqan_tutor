**Goal:** A video library (uploaded MP4 ≤ 20 MB or YouTube) on the public site, audio and video testimonials,
and an optional intro video — behind `video_library` and `testimonial_media`, off by default.

testimonials and home-page forms.

**Produces:** `Video` (D-4, `Source` choices `UPLOAD="upload"`, `YOUTUBE="youtube"`, check constraint),
`Testimonial.kind` (`Kind` `TEXT`/`AUDIO`/`VIDEO`, default text), `Testimonial.audio`, `Testimonial.video`

```python
import re
from urllib.parse import parse_qs, urlsplit

from etqan.platform.exceptions import ValidationError

ID_RE = re.compile(r"^[A-Za-z0-9_-]{11}$")
HOSTS = {"youtube.com", "www.youtube.com", "m.youtube.com"}
SHORT = "youtu.be"
PREFIXES = ("/shorts/", "/embed/", "/live/")
MESSAGE = "Paste a YouTube link or video id."


def parse_id(raw: str) -> str:
    """D-2: the 11-character id of a YouTube video, from a bare id or a link."""
    text = (raw or "").strip()
    if ID_RE.fullmatch(text):
        return text
    parts = urlsplit(text)
    host = (parts.hostname or "").lower()
    candidate = ""
    if parts.scheme in ("http", "https"):
        if host == SHORT:
            candidate = parts.path.strip("/").split("/")[0]
        elif host in HOSTS:
            if parts.path == "/watch":
                candidate = parse_qs(parts.query).get("v", [""])[0]
            else:
                for prefix in PREFIXES:
                    if parts.path.startswith(prefix):
                        candidate = parts.path[len(prefix):].split("/")[0]
    if ID_RE.fullmatch(candidate):
        return candidate
    raise ValidationError(MESSAGE, field="external_id")
```

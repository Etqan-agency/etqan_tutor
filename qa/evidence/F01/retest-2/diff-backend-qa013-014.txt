"""The third-party admins on the Etqan console, with their actions behind the
change permission (ETQAN-QA-013).

Django offers an action to anyone who may view the list unless the action
names its permissions. Etqan's own admins say `permissions=["change"]` on
every action that changes data or sends something; django-celery-beat's
task actions (enable, disable, toggle, run) and allauth's "make verified"
don't, so a view-only staff member could run them. They are registered
again here with the same rule. Admin autodiscovery imports this module after
theirs (`etqan.platform` follows both apps in INSTALLED_APPS)."""

from allauth.account.admin import EmailAddressAdmin
from allauth.account.models import EmailAddress
from django.contrib import admin
from django_celery_beat.admin import PeriodicTaskAdmin
from django_celery_beat.models import PeriodicTask


class ChangeToRunActions:
    """Only a staff member who may change the model is offered its actions
    (delete keeps its own delete permission). Django validates the posted
    action against `get_actions`, so a forged one is refused too."""

    def get_actions(self, request):
        actions = super().get_actions(request)
        if self.has_change_permission(request):
            return actions
        return {name: a for name, a in actions.items() if name == "delete_selected"}


class EtqanPeriodicTaskAdmin(ChangeToRunActions, PeriodicTaskAdmin):
    pass


class EtqanEmailAddressAdmin(ChangeToRunActions, EmailAddressAdmin):
    pass


for model, model_admin in (
    (PeriodicTask, EtqanPeriodicTaskAdmin),
    (EmailAddress, EtqanEmailAddressAdmin),
):
    admin.site.unregister(model)
    admin.site.register(model, model_admin)

elsewhere (suspended, its defaults suspended, a feature switched off by
`set_features`) writes only what its user changed on the page; the newer
values of everything else stay."""

from html.parser import HTMLParser

class _Form(HTMLParser):
    """What a browser posts for the page's academy form: named inputs, a
    checked box as "on", a select's chosen option, a textarea's text."""

    def __init__(self):
        super().__init__()
        self.data: dict[str, str] = {}
        self._inside = False
        self._select: str | None = None
        self._textarea: str | None = None

    def handle_starttag(self, tag, attrs):
        attr = dict(attrs)
        if tag == "form":
            self._inside = attr.get("id") == "academy_form"
        if not self._inside:
            return
        if tag == "option":
            if self._select and "selected" in attr:
                self.data[self._select] = attr.get("value") or ""
        elif tag == "input" and "name" in attr:
            self._input(attr)
        elif tag == "select" and "name" in attr:
            self._select = attr["name"]
        elif tag == "textarea" and "name" in attr:
            self._textarea = attr["name"]
            self.data[attr["name"]] = ""

    def _input(self, attr):
        kind = attr.get("type", "text")
        if kind == "submit" or (kind == "checkbox" and "checked" not in attr):
            return
        default = "on" if kind == "checkbox" else ""
        self.data[attr["name"]] = attr.get("value") or default

    def handle_endtag(self, tag):
        if tag == "form":
            self._inside = False
        elif tag == "select":
            self._select = None
        elif tag == "textarea":
            self._textarea = None

    def handle_data(self, data):
        if self._textarea:
            self.data[self._textarea] += data.removeprefix("\n")


def page_data(resp) -> dict:
    parser = _Form()
    parser.feed(resp.content.decode())
    return parser.data
